## Installation

```bash
npx shadcn@latest add @preskok/checkbox
```

## Usage

```tsx
import {
  Checkbox,
  CheckboxGroup,
} from "@/components/ui/preskok-ui/checkbox"

export function CheckboxGroupExample() {
  return (
    <CheckboxGroup label="Select interests">
      <Checkbox value="dev">Development</Checkbox>
      <Checkbox value="design">Design</Checkbox>
      <Checkbox value="pm">Product Management</Checkbox>
    </CheckboxGroup>
  )
}
```
