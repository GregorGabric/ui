## Installation

```bash
npx shadcn@latest add @preskok/experimental-pie-chart
```

## Usage

```tsx
import { ExperimentalPieChart } from "@/components/ui/preskok-ui/experimental-pie-chart"

;<ExperimentalPieChart
  config={{
    chrome: { label: "Chrome", color: "var(--chart-1)" },
  }}
  data={data}
  dataKey="visitors"
  nameKey="browser"
/>
```
