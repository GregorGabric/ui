## Installation

```bash
npx shadcn@latest add @preskok/experimental-area-chart
```

## Usage

```tsx
import { ExperimentalAreaChart } from "@/components/ui/preskok-ui/experimental-area-chart"

;<ExperimentalAreaChart
  config={{
    desktop: { label: "Desktop", color: "var(--chart-1)" },
  }}
  data={data}
  dataKey="month"
  fillType="gradient"
  lineType="natural"
/>
```
