## Installation

```bash
npx shadcn@latest add @preskok/experimental-line-chart
```

## Usage

```tsx
import { ExperimentalLineChart } from "@/components/ui/preskok-ui/experimental-line-chart"

;<ExperimentalLineChart
  config={{
    desktop: { label: "Desktop", color: "var(--chart-1)" },
  }}
  data={data}
  dataKey="month"
  lineType="natural"
/>
```
