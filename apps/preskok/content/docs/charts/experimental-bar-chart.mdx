## Installation

```bash
npx shadcn@latest add @preskok/experimental-bar-chart
```

## Usage

```tsx
import { ExperimentalBarChart } from "@/components/ui/preskok-ui/experimental-bar-chart"

;<ExperimentalBarChart
  config={{
    desktop: { label: "Desktop", color: "var(--chart-1)" },
    mobile: { label: "Mobile", color: "var(--chart-2)" },
  }}
  data={data}
  dataKey="month"
/>
```
