## Installation

```bash
npx shadcn@latest add @preskok/experimental-radial-chart
```

## Usage

```tsx
import { ExperimentalRadialChart } from "@/components/ui/preskok-ui/experimental-radial-chart"

;<ExperimentalRadialChart
  data={data}
  dataKey="visitors"
  nameKey="browser"
  config={{
    chrome: { label: "Chrome", color: "var(--chart-1)" },
  }}
/>
```
