## Usage

```tsx
import { LineChart } from "@/components/ui/preskok-ui/line-chart"

export function Example() {
  return (
    <LineChart
      dataKey="month"
      data={[]}
      config={{ value: { label: "Value" } }}
    />
  )
}
```
