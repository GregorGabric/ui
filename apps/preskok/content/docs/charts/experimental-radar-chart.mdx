## Installation

```bash
npx shadcn@latest add @preskok/experimental-radar-chart
```

## Usage

```tsx
import { ExperimentalRadarChart } from "@/components/ui/preskok-ui/experimental-radar-chart"

;<ExperimentalRadarChart
  data={data}
  dataKey="month"
  config={{
    desktop: { label: "Desktop", color: "var(--chart-1)" },
  }}
/>
```
