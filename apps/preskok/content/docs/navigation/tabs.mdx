## Usage

```tsx
import { Tabs } from "@/components/ui/preskok-ui/tabs"

export function TabsDemo() {
  return (
    <Tabs defaultSelectedKey="1" items={[{ id: "1", title: "Tab 1" }]}>
      {(item) => <div>{item.title}</div>}
    </Tabs>
  )
}
```
