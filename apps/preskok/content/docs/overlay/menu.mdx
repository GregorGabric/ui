## Usage

```tsx
import { Button } from "@/components/ui/preskok-ui/button"
import {
  Menu,
  MenuContent,
  MenuItem,
  MenuSection,
} from "@/components/ui/preskok-ui/menu"

export function MenuDemo() {
  return (
    <Menu>
      <Button intent="outline">Actions</Button>
      <MenuContent>
        <MenuSection label="File">
          <MenuItem id="new">New</MenuItem>
        </MenuSection>
      </MenuContent>
    </Menu>
  )
}
```
