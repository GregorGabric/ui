## Installation

```bash
npx preskok-ui@latest add dropdown
```

## Usage

```tsx
import { Button } from "@/components/ui/preskok-ui/button"
import {
  DropdownDescription,
  DropdownLabel,
} from "@/components/ui/preskok-ui/dropdown"
import {
  Menu,
  MenuContent,
  MenuItem,
  MenuTrigger,
} from "@/components/ui/preskok-ui/menu"

export function Example() {
  return (
    <Menu>
      <MenuTrigger>
        <Button intent="outline">Open</Button>
      </MenuTrigger>
      <MenuContent>
        <MenuItem id="profile">
          <DropdownLabel>Profile</DropdownLabel>
          <DropdownDescription>View profile</DropdownDescription>
        </MenuItem>
      </MenuContent>
    </Menu>
  )
}
```

## Components

- **DropdownItem**, **DropdownLabel**, **DropdownDescription**, **DropdownSection**, **DropdownSeparator**, **DropdownKeyboard**
