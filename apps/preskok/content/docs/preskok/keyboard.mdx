## Installation

```bash
npx preskok-ui@latest add keyboard
```

## Usage

```tsx
import { Keyboard } from "@/registry/preskok/ui/preskok-ui/keyboard"

export function Example() {
  return <Keyboard keys={["⌘", "K"]} />
}
```
