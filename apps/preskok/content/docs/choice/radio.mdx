## Usage

```tsx
import { Radio, RadioGroup } from "@/registry/preskok/ui/preskok-ui/radio"

export function RadioDemo() {
  return (
    <RadioGroup label="Plan" defaultValue="basic">
      <Radio value="basic">Basic</Radio>
      <Radio value="pro">Pro</Radio>
    </RadioGroup>
  )
}
```
