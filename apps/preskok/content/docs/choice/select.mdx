## Usage

```tsx
import { Select } from "@/components/ui/preskok-ui/select"

export function SelectDemo() {
  return (
    <Select
      label="Project"
      defaultSelectedKey="1"
      items={[{ id: "1", name: "Alpha" }]}
    >
      {(item) => <span>{item.name}</span>}
    </Select>
  )
}
```
