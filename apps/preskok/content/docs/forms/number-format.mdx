## Usage

```tsx
import {
  NumberFormatProvider,
  NumberText,
  useNumberFormat,
} from "@/components/ui/preskok-ui/number-format/number-format"

export function NumberFormatDemo() {
  return (
    <NumberFormatProvider locale="en-US" currency="USD">
      <NumberText value={12345.67} />
    </NumberFormatProvider>
  )
}
```
