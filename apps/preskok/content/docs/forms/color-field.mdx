## Installation

```bash
npx shadcn@latest add @preskok/color-field
```

## Usage

```tsx
import { ColorField } from "@/components/ui/preskok-ui/color-field"

export function ColorFieldExample() {
  return (
    <ColorField
      label="Brand color"
      placeholder="#2563eb"
      defaultValue="#2563eb"
      description="Choose a hex color."
    />
  )
}
```
