## Installation

```bash

```

## Usage

```tsx
import { ListBox, ListBoxItem } from "@/components/ui/preskok-ui/list-box"

export function Example() {
  return (
    <ListBox aria-label="Options">
      <ListBoxItem id="a">A</ListBoxItem>
      <ListBoxItem id="b">B</ListBoxItem>
    </ListBox>
  )
}
```
