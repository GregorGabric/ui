## Installation

```bash
npx preskok-ui@latest add disclosure
```

## Usage

```tsx
import {
  Disclosure,
  DisclosureGroup,
  DisclosurePanel,
  DisclosureTrigger,
} from "@/components/ui/preskok-ui/disclosure"

export function Example() {
  return (
    <DisclosureGroup>
      <Disclosure>
        <DisclosureTrigger>Section</DisclosureTrigger>
        <DisclosurePanel>Content</DisclosurePanel>
      </Disclosure>
    </DisclosureGroup>
  )
}
```
