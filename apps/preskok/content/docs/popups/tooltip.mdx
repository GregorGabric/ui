## Usage

```tsx
import {
  Tooltip,
  TooltipContent,
  TooltipTrigger,
} from "@/components/ui/preskok-ui/tooltip"

export function TooltipDemo() {
  return (
    <Tooltip>
      <TooltipTrigger aria-label="mileage">Mileage ?</TooltipTrigger>
      <TooltipContent>34 mpg</TooltipContent>
    </Tooltip>
  )
}
```
