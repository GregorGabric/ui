## Usage

```tsx
import { ScrollArea } from "@/registry/preskok/ui/preskok-ui/scroll-area"

export function ScrollAreaDemo() {
  return (
    <ScrollArea className="h-48 w-64">
      <div className="space-y-2">
        {Array.from({ length: 50 }, (_, i) => (
          <div key={i}>Item {i + 1}</div>
        ))}
      </div>
    </ScrollArea>
  )
}
```
