## Installation

```bash
npx shadcn@latest add @preskok/grid-list
```

## Usage

```tsx
import {
  GridList,
  GridListItem,
} from "@/components/ui/preskok-ui/grid-list"

export function Example() {
  return (
    <GridList aria-label="Items">
      <GridListItem id="a">A</GridListItem>
      <GridListItem id="b">B</GridListItem>
    </GridList>
  )
}
```
