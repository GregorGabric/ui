## Installation

```bash
npx preskok-ui@latest add container
```

## Usage

```tsx
import { Container } from "@/registry/preskok/ui/preskok-ui/container"

export function ContainerDemo() {
  return (
    <>
      <Container className="bg-muted p-4">Default padding</Container>
      <Container constrained className="bg-muted mt-2 p-4">
        Constrained padding
      </Container>
    </>
  )
}
```
