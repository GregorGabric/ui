## Usage

```tsx
import { toast } from "sonner"

import { Toast } from "@/components/ui/preskok-ui/toast"

export function ToastDemo() {
  return (
    <div>
      <button onClick={() => toast.success("Saved!")}>Show toast</button>
      <Toast />
    </div>
  )
}
```
