## Usage

```tsx
import { toast } from "sonner"

import { Toast } from "@/registry/preskok/ui/preskok-ui/toast"

export function ToastDemo() {
  return (
    <div>
      <button onClick={() => toast.success("Saved!")}>Show toast</button>
      <Toast />
    </div>
  )
}
```
