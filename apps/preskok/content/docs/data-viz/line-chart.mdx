## Installation

```bash
npx preskok-ui@latest add line-chart
```

## Usage

```tsx
import { LineChart } from "@/registry/preskok/ui/preskok-ui/line-chart"

export function Example() {
  return (
    <LineChart
      dataKey="month"
      data={[]}
      config={{ value: { label: "Value" } }}
    />
  )
}
```
