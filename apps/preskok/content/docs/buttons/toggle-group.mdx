## Usage

```tsx
import { ToggleGroup } from "@/components/ui/preskok-ui/toggle-group"

export function ToggleGroupDemo() {
  return (
    <ToggleGroup selectionMode="single" aria-label="Formatting">
      <ToggleGroup.Item id="bold">Bold</ToggleGroup.Item>
      <ToggleGroup.Item id="italic">Italic</ToggleGroup.Item>
      <ToggleGroup.Item id="underline">Underline</ToggleGroup.Item>
    </ToggleGroup>
  )
}
```
