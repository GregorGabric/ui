## Installation

```bash
npx preskok-ui@latest add link
```

## Usage

```tsx
import { Link } from "@/registry/preskok/ui/preskok-ui/link"

export function Example() {
  return (
    <Link href="#" intent="primary">
      Docs
    </Link>
  )
}
```
